def storage():

            break

            print("to nemas v inventari")
            continue

            enemy_health[0] -= 20
            damage_storage.clear()
            damage_storage.append(20)
            cyclus()
            print("bodnul jsi nozem")
            print(newline)
            smrt()
            break

        elif choice == "armor":
            armor.append("w")
            inventar.remove("armor")
            print("armor equiped")
            print(newline)
            smrt()
            break

            health.clear()
            health.append(100)
            inventar.remove("potion")
            print("potion vypitej")
            print(newline)
            smrt()
            break

        elif choice == "mec":
            enemy_health[0] -= 25
            damage_storage.clear()
            damage_storage.append(25)
            print("seknul jsi mecem")
            cyclus()
            smrt()
            break

        elif choice == "shield":
            shield.append("w")
            inventar.remove("shield")
            print("dal  jsi shield na ruku a jdes se s nim branit")
            print(newline)
            smrt()

#thole je kdyz se dostanes do boje tak aby jsi si vybral co udelas
def fight():
    while enemy_health[0] > 0:
        while True:
            print(newline)
            press=input("Co chces udelat? F pro fight, I inventori: ").upper()
            if press == "F":
                utok()
                print("dal jsi uder pesti")
                print(newline)
                break
            elif press == "I":
                pre_storage()
                break
            else:
                print("blby zvoleni")

        if damage_storage:
            print(f"tvuj health {health[0]}")
            print(f"tvuj damage tohle kolo {damage_storage[0]}")
            print(f"enemy health {max(enemy_health[0], 0)}")
            damage_storage.clear()

    smrt()

    
        
#tohle kontroluje kolik dostanes damage podhle toho co jsi si dal na sebe nebo pouzil
def cyclus():
    if shield and armor:
        damage = random.choice([0, 0, 3, 4])
        health[0] -= damage
        smrt()
    elif shield:
        damage = random.choice([1, 7, 2, 3, 4, 5, 6, 0, 0, 0, 0, 0, 0, 0])
        health[0] -= damage
        smrt()
    elif armor:
        damage = random.randint(1, 4)
        health[0] -= damage
        smrt()
    else:
        damage = random.randint(1, 7)
        health[0] -= damage
        print(newline)
        smrt()



#tohle se stane kdyz ve fight das f tak enemy dostane damage

    smrt()
